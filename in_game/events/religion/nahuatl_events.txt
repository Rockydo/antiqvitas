nahuatl_events.2 = { #Precious Son
	hide_portraits = yes
	type = country_event
	title = nahuatl_events.2.title
	desc = nahuatl_events.2.desc
	outcome = neutral

	trigger = {
		current_date > 476.9.4
		religion = religion:nahuatl
		NOT = {
			has_variable = nahuatl_warrior_born
		}
	}

	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
		create_character = {
			estate = estate_type:peasants_estate
			culture = root.culture
			religion = root.religion
			adm = { 0 20 }
			dip = { 0 20 }
			mil = { 20 50 }
			age = 0
			save_scope_as = child_scope
		}
		set_variable = { name = nahuatl_warrior_born value = scope:child_scope }
	}

	option = { #https://en.wikipedia.org/wiki/T%C4%93lpochcalli
		name = nahuatl_events.2.a
		trigger_event_silently = { id = nahuatl_events.3 years = 16 }
		custom_tooltip = nahuatl_events.2.a.tt
	}
	option = { #https://en.wikipedia.org/wiki/Calmecac
		name = nahuatl_events.2.b
		set_variable = { name = nahuatl_warrior_calmecac value = 1 }
		add_government_power = government_power_extreme_penalty
		custom_tooltip = nahuatl_events.2.b.tt
