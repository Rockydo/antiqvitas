on_chinese_expedition_location_move = {
	if = {
		limit = {
			scope:expedition_initiator ?= {
				has_variable = chinese_treasure_fleet_location
				NOT = {
					has_discovered = scope:expedition_initiator.var:chinese_treasure_fleet_location
				}
			}
		}
		scope:expedition_initiator = {
			add_to_variable_list = {
				name = list_of_discovered_locations
				target = var:chinese_treasure_fleet_location
			}
		}
	}
}

set_patriarch_variables = {
	$organization$ = {
		set_variable = {
			name = patriarch
			value = $character$
		}
	}
	$character$ = {
		set_variable = {
			name = is_patriarch
			value = $organization$
		}
	}
}

set_pentarch_variable = { #Character scope, $location$ is the pentarchy seat
	set_variable = {
		name = is_pentarch
		value = $location$
	}
}

select_suitable_patriarch = {
	if = {
		limit = {
			any_character = {
				is_adult = yes
				is_female = no
				has_estate = estate_type:clergy_estate
				religion = prev.religion
				is_artist = no
				trigger_if = {
					limit = { $pentarchy$ = yes }
					trigger_if = {
						limit = {
							prev = {
								is_leader_of_international_organization_of_type = { type = autocephalous_patriarchate }
							}
						}
						has_variable = is_patriarch
					}
					NOT = { has_variable = is_pentarch }
				}
				trigger_else = {
					NOT = { has_variable = is_patriarch }
				}
			}
		}
		random_character = {
			limit = {
				is_adult = yes
				is_female = no
				has_estate = estate_type:clergy_estate
				religion = prev.religion
				is_artist = no
				trigger_if = {
					limit = { $pentarchy$ = yes }
					trigger_if = {
						limit = {
							prev = {
								is_leader_of_international_organization_of_type = { type = autocephalous_patriarchate }
							}
						}
						has_variable = is_patriarch
					}
					NOT = { has_variable = is_pentarch }
				}
				trigger_else = {
					NOT = { has_variable = is_patriarch }
				}
			}
			save_scope_as = target_patriarch
		}
	}
	else = {
		create_character = {
			religion = this.religion
			estate = estate_type:clergy_estate
			age = { 40 50 }
			save_scope_as = target_patriarch
		}
	}
}

#$organization$ = autocephalous patriarchate IO, $leader$ = leader_country
set_up_patriarch = {
	$leader$ = {
		select_suitable_patriarch = { pentarchy = no }
	}
	set_patriarch_variables = { organization = $organization$ character = scope:target_patriarch }
}

setup_area_preferences = {
	every_country = {
		limit = { is_ai = yes}
		add_area_preference = avoid_siberia_colonization
		#American / African natives should be allowed to colonize their own continent
		if = {
			limit = { exists = capital NOT = { capital.continent = continent:america } }
			add_area_preference = avoid_american_colonization
		}
		if = {
			limit = { exists = capital NOT = { capital.continent = continent:africa } }
			add_area_preference = avoid_african_colonization
		}
		if = {
			limit = { exists = capital NOT = { capital.continent = continent:oceania } }
			add_area_preference = avoid_oceania_colonization
		}
		if = {
			limit = { exists = capital NOT = { capital.sub_continent = sub_continent:south_east_asia } }
			add_area_preference = avoid_south_east_asica_colonization
		}
	}
	region:japan_region = {
		every_country_with_capital_in_geography = {
			add_area_preference = generic_conquer_home_region
		}
	}
	c:MOS ?= {
		remove_area_preference = avoid_siberia_colonization
	}
	c:RUS ?= {
		remove_area_preference = avoid_siberia_colonization
		remove_area_preference = avoid_american_colonization	#relevant for Alaska
