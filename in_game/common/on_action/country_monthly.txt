on_papal_opinion_added = {
	trigger = {
		OR = {
			modifier:papal_relations < 0
			modifier:papal_relations > 0
		}
		country_exists = c:DUMMY
		not = { tag = PAP }
	}
	effect = {
		c:DUMMY = {
			remove_opinion = {
				target = prev
				modifier = opinion_papal_relations
			}
		}
		reverse_add_opinion = {
			target = c:DUMMY
			modifier = opinion_papal_relations
			scale = modifier:papal_relations
		}
	}
}

on_papal_opinion_removed = {
	trigger = {
		country_exists = c:DUMMY
		not = { tag = PAP }
		c:DUMMY = {
			has_opinion = {
				target = prev
				modifier = opinion_papal_relations
			}
		}
		modifier:papal_relations = 0
	}
	effect = {
		c:DUMMY = {
			remove_opinion = {
				target = prev
				modifier = opinion_papal_relations
			}
		}
	}
}

on_papal_authority_updated = {
	trigger = {
		tag = PAP
		country_exists = c:DUMMY
		is_member_of_international_organization = international_organization:catholic_church
	}
	effect = {
		if = {
			limit = {
				international_organization:catholic_church = {
					var:papal_authority >= 75
				}
			}
			religion:catholic = {
				add_religion_modifier = {
					modifier = high_papal_authority
					days = -1
				}
				remove_religion_modifier = low_papal_authority
			}
		}
		else_if = {
			limit = {
				international_organization:catholic_church = {
					var:papal_authority < 25
				}
			}
			religion:catholic = {
				add_religion_modifier = {
					modifier = low_papal_authority
					days = -1
				}
				remove_religion_modifier = high_papal_authority
			}
		}
		else = {
			religion:catholic = {
				remove_religion_modifier = high_papal_authority
				remove_religion_modifier = low_papal_authority
			}
		}
	}
}

on_fraticide_underage_succession_waiting = {
	trigger = {
		has_variable = fraticide_succession_has_child_sons
	}
	effect = {
		if = {
			limit = {
				list_size = { name = rival_male_children value > 0 }
			}
			if = {
				limit = {
					any_in_list = {
						list = rival_male_children
						is_alive = yes
						is_adult = yes
					}
				}
				trigger_event_silently = succession.5
			}
			else = {
				every_in_list = {
					list = rival_male_children
					limit = { is_alive = no }
					remove_from_list = rival_male_children
				}
			}
		}
		else = {
			trigger_event_silently = succession.5
		}
	}
}
# ANTIQVITAS M6: deterministic sourced Julio-Claudian succession deadlines.
antq_m6_roman_succession_monthly = {
	trigger = { tag = XAA }
	effect = {
		if = {
			limit = {
				current_date > 4.2.20
				has_variable = antq_m6_roman_gaius_caesar
				has_variable = antq_m6_roman_tiberius
			}
			trigger_event_silently = antq_m6.2
		}
		if = {
			limit = {
				current_date > 14.8.18
				has_variable = antq_m6_roman_augustus
				has_variable = antq_m6_roman_tiberius
			}
			trigger_event_silently = antq_m6.3
		}
		if = {
			limit = {
				current_date > 37.3.15
				has_variable = antq_m6_roman_tiberius
			}
			var:antq_m6_roman_tiberius = {
				remove_character_modifier = antq_m6_historical_lifespan_guard
			}
			remove_variable = antq_m6_roman_tiberius
		}
		if = {
			limit = {
				has_variable = antq_m6_roman_succession_reserve
				ruler ?= var:antq_m6_roman_succession_reserve
			}
			var:antq_m6_roman_succession_reserve = {
				remove_character_modifier = antq_m6_historical_lifespan_guard
			}
			remove_variable = antq_m6_roman_succession_reserve
			create_character = {
				culture = culture:antq_latin
				religion = religion:antq_religio_romana
				dynasty = dynasty:antq_julio_claudian_dynasty
				estate = estate_type:nobles_estate
				age = { 20 30 }
				save_scope_as = antq_m6_roman_succession_reserve_scope
			}
			scope:antq_m6_roman_succession_reserve_scope = {
				add_character_modifier = { modifier = antq_m6_historical_lifespan_guard years = -1 mode = add_and_extend }
			}
			set_variable = { name = antq_m6_roman_succession_reserve value = scope:antq_m6_roman_succession_reserve_scope }
			set_as_designated_heir = scope:antq_m6_roman_succession_reserve_scope
		}
		# Seat empty cabinet slots after courtiers die so the 0.20
		# crown-from-cabinet lever does not go unused. The first-open
		# effect no-ops when every slot is already filled.
		random_character = {
			limit = { is_adult = yes in_cabinet = no NOT = { is_heir = yes } }
			save_scope_as = antq_m6_cabinet_refill_1
		}
		if = {
			limit = { exists = scope:antq_m6_cabinet_refill_1 }
			add_character_to_first_open_cabinet = { target = scope:antq_m6_cabinet_refill_1 }
		}
		random_character = {
			limit = { is_adult = yes in_cabinet = no NOT = { is_heir = yes } }
			save_scope_as = antq_m6_cabinet_refill_2
		}
		if = {
			limit = { exists = scope:antq_m6_cabinet_refill_2 }
			add_character_to_first_open_cabinet = { target = scope:antq_m6_cabinet_refill_2 }
		}
		random_character = {
			limit = { is_adult = yes in_cabinet = no NOT = { is_heir = yes } }
			save_scope_as = antq_m6_cabinet_refill_3
		}
		if = {
			limit = { exists = scope:antq_m6_cabinet_refill_3 }
			add_character_to_first_open_cabinet = { target = scope:antq_m6_cabinet_refill_3 }
		}
	}
}

# ANTIQVITAS: standing ancient councils reset the lack-of-parliament penalty.
antq_ancient_council_session = {
	trigger = {
		months_since_last_parliament_called > 36
		is_active_parliament = no
	}
	effect = {
		start_parliament_effect = yes
		end_parliament = yes
	}
}
